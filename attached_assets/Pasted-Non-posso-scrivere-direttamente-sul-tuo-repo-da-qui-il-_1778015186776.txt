## 5. Bottom navigation mobile

**Obiettivo:** aggiungere una bottom‑nav fissa su mobile, montata nel layout pubblico, che usi le stesse rotte e icone della navbar. 

### 5.1. Crea `bottom-nav.tsx`

**Nuovo file:**  
`artifacts/orientamento/src/components/layout/bottom-nav.tsx`

Esempio base (adatta import in base alla tua struttura):

```tsx
import { Home, FlaskConical, Layers, BookOpenText, User } from "lucide-react";
import { useLocation, Link } from "wouter";
import { useAuth } from "@/contexts/AuthContext";
import { cn } from "@/lib/utils";

const NAV_ITEMS = [
  { href: "/dashboard", label: "Dashboard", icon: Home },
  { href: "/test", label: "Test", icon: FlaskConical },
  { href: "/settori", label: "Settori", icon: Layers },
  { href: "/crescita", label: "Crescita", icon: BookOpenText },
  { href: "/profilo", label: "Profilo", icon: User },
];

export function BottomNav() {
  const { isLoggedIn } = useAuth();
  const [location] = useLocation();

  if (!isLoggedIn) return null;

  return (
    <nav className="md:hidden fixed bottom-0 left-0 right-0 z-50 border-t border-border bg-background/90 backdrop-blur px-2 pb-[calc(env(safe-area-inset-bottom,0px)+8px)] pt-1.5">
      <div className="flex items-center justify-between gap-1 max-w-5xl mx-auto">
        {NAV_ITEMS.map(({ href, label, icon: Icon }) => {
          const active = location === href;
          return (
            <Link
              key={href}
              href={href}
              className={cn(
                "flex flex-col items-center justify-center flex-1 py-1.5 rounded-xl text-[11px] font-medium transition-colors",
                active ? "text-primary" : "text-muted-foreground"
              )}
            >
              <Icon
                className={cn(
                  "w-5 h-5 mb-0.5",
                  active ? "text-primary" : "text-muted-foreground"
                )}
              />
              <span className="truncate">{label}</span>
            </Link>
          );
        })}
      </div>
    </nav>
  );
}
```

### 5.2. Monta in `App.tsx`

**File:** `artifacts/orientamento/src/App.tsx` 

- Aggiungi l’import:

```ts
import { BottomNav } from "@/components/layout/bottom-nav";
```

- Nel layout pubblico, dopo `<Footer />`, aggiungi `<BottomNav />`:

```tsx
<Route>
  <div className="flex flex-col min-h-[100dvh]">
    <Navbar />
    <main className="flex-1">
      <BackButton />
      <AnimatedRoutes />
    </main>
    <Footer />
    <BottomNav /> {/* nuovo */}
  </div>
</Route>
```

Se vedi che la bottom‑nav copre il contenuto, puoi aggiungere un po’ di padding‑bottom al `<main>` su mobile.

## 6. `PremiumGate` riutilizzabile

**Nuovo file:**  
`artifacts/orientamento/src/components/PremiumGate.tsx`

Esempio:

```tsx
import { Crown } from "lucide-react";
import { Link } from "wouter";
import { useAuth } from "@/contexts/AuthContext";
import { cn } from "@/lib/utils";

interface PremiumGateProps {
  children: React.ReactNode;
  featureName: string; // es: "Wiki AI"
}

export function PremiumGate({ children, featureName }: PremiumGateProps) {
  const { user } = useAuth();
  const isPremium = (user as any)?.isPremium === true;

  if (isPremium) return <>{children}</>;

  return (
    <div className="relative rounded-2xl border bg-card overflow-hidden">
      <div className="opacity-40 pointer-events-none select-none">
        {children}
      </div>
      <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 bg-background/75 backdrop-blur-sm p-6">
        <div className="flex items-center gap-2">
          <Crown className="w-5 h-5 text-primary" />
          <span className="font-serif text-lg font-semibold">
            Funzionalità Premium
          </span>
        </div>
        <p className="text-sm text-muted-foreground text-center max-w-sm">
          Sblocca <span className="font-medium">{featureName}</span> con il piano Premium e ottieni un’analisi più approfondita.
        </p>
        <Link
          href="/premium"
          className={cn(
            "inline-flex items-center justify-center rounded-full px-5 py-2 text-sm font-medium",
            "bg-primary text-primary-foreground hover:bg-primary/90 transition-colors"
          )}
        >
          Passa a Premium
        </Link>
      </div>
    </div>
  );
}
```

Poi, nelle pagine AI (es. `wiki.tsx`, `roadmap.tsx`, `colloquio.tsx`, `skills-gap.tsx`, `coach.tsx`), avvolgi il contenuto “premium” in:

```tsx
<PremiumGate featureName="Wiki AI">
  {/* contenuto esistente */}
</PremiumGate>
```
